\section{Background}
\begin{frame}{PMOS LDO Schematic}
    \svg{../images/pmos_ldo.svg}{General LDO schematic}{1}

\begin{frame}{LDO's Small-Signal Model}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.4\textwidth}
            \small
            \textbf{Output voltage}
            \begin{equation*}
                \frac{V_{OUT}}{V_{REF}} \approx \frac{1}{\beta},
                \qquad
                \frac{R_1}{R_2} = \frac{V_{REF}}{V_{OUT}} - 1
            \end{equation*}
            \textbf{PSRR}
            \begin{equation*}
                |PSRR_{DC}| \cong
                \frac{1}{\beta A_{EA,0}g_{m,PD}r_{ds,p}}
            \end{equation*}
        \end{column}
        \begin{column}{0.6\textwidth}
            \small
            \textbf{Stability}
            \begin{equation*}
                \omega_{p1}=\frac{1}{g_{m,PD}R_1R_2C_c},
                \quad
                \omega_{p2}=\frac{g_{m,PD}}{C_1+C_2},
                \quad
                \omega_{z}=\frac{-1}{C_c(1 \slash g_{m,PD}-R_c)}
            \end{equation*}
            \textbf{Output-referred noise}
            \begin{equation*}
                S_{n,o}(f) \cong (S_{n,ref}(f)+S_{n,EA}(f))(1+\frac{R_1}{R_2})^2
                +S_{n,R2}(f)(\frac{R_1}{R_2})^2+S_{n,R_1}(f)
            \end{equation*}
        \end{column}
    \end{columns}
    \vspace{-0.3em}
    \begin{figure}
        \centering
        \includesvg[width=0.72\linewidth,height=0.40\textheight,keepaspectratio]{../images/pmos_ldo_ss.svg}
        \caption{Small-signal model of the LDO}
    \end{figure}
\end{frame}

\section{5T OTA-Based LDO}
\begin{frame}{5T OTA (Simple Differential Pair)}
    \begin{columns}[c,onlytextwidth]
        \begin{column}{0.48\textwidth}
            The 5T OTA is a simple differential pair with a gain of:
            \begin{equation*}
                A_{5T}=\frac{g_{m,12}}{g_{5T,out}}
            \end{equation*}
            while the output impedance is given by:
            \begin{equation*}
                g_{5T,out}=g_{ds,2}+g_{ds,4}
            \end{equation*}
            As for the output swing, to keep all transistors in saturation, we must have:
            \begin{equation*}
                V_{REF} - V_{T} \leq V_{OUT} \leq V_{EXT} - V_{DSAT}
            \end{equation*}
        \end{column}
        \begin{column}{0.48\textwidth}
            \svg{../images/5t_ota.svg}{5T OTA}{1}
        \end{column}
    \end{columns}
\end{frame}

\section{Telescopic OTA-Based LDO}
\begin{frame}{Telescopic OTA}
    \svg{../images/tel_ota.svg}{Telescopic OTA}{1}
\end{frame}
\begin{frame}{Telescopic OTA}
    \begin{columns}[c,onlytextwidth]
        \begin{column}{0.48\textwidth}
            \svg{../images/tel_ota.svg}{Telescopic OTA}{1}
            This leads to the following output swing:
            \begin{equation*}
                V_{REF} - V_{T} + V_{DSAT} \leq V_{OUT} \leq V_{EXT} - 2V_{DSAT}
            \end{equation*}
        \end{column}
        \begin{column}{0.48\textwidth}
            Both loop gain and output impedance are substantially increased compared to the 5T OTA. More specifically, the loop gain can be given by:
            \begin{equation*}
                A_{TEL} = \frac{g_{m,12}}{g_{tel,out}}
            \end{equation*}
            whereas the output impedance is given by:
            \begin{equation*}
                g_{tel,out} =
                    g_{ds,2} \frac{g_{ds,2C}}{g_{m,2C}} +
                    g_{ds,4} \frac{g_{ds,4C}}{g_{m,4C}}
            \end{equation*}
            Bias transistors should be sized as follows to ensure maximum output swing:
            \begin{equation*}
                \frac{W_{8}}{L_{8}} = \frac{W_{3,4}}{4L_{3,4}},
                \quad
                \frac{W_{10}}{L_{10}} = \frac{W_{1,2}}{4L_{1,2}}
            \end{equation*}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Telescopic OTA-Based LDO (Sizing)}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.55\textwidth}
            \begin{itemize}
                \item \textbf{Input Pair \& Active Load:} Large W and L to reduce flicker noise and increase loop gain.
                \item \textbf{Cascode Transistors:} Small W to ensure saturation; no need for larger transconductance.
                \item \textbf{Current Mirror:} Large L for better current mirroring; small W to keep all devices in saturation while improving PSRR.
                \item \textbf{Pass Device:} Large W to improve transconductance and loop gain.
                \item \textbf{Divider:} Small resistors to minimize noise and decrease the pass device's gate voltage at no-load.
                \item \textbf{Compensation:} Nulling resistor and compensation capacitor for improving UGBW and PSRR at higher frequencies.

\begin{frame}{Comparison}
    \begin{itemize}
        \item \textbf{5T OTA-Based LDO:} Offers the most balanced results. Should be preferred when:
        \begin{enumerate}
            \item A smaller pass device is needed.
            \item Better error amplifier output swing is required to keep all devices in saturation.
            \item A simpler design with fewer transistors is desired.
            \item Slightly better noise performance at high frequencies is needed.
            \item Slightly better PSRR performance at high frequencies is needed.
        \end{enumerate}
        \item \textbf{Telescopic OTA-Based LDO:} Maximizes loop gain and PSRR. Should be preferred when:
        \begin{enumerate}
            \item A larger pass device can be utilized.
            \item The limited output swing of the error amplifier is not a concern.
            \item More complex designs with additional transistors are acceptable.
            \item Higher PSRR values at low frequencies are required.
            \item Better phase margin under load is needed.
        \end{enumerate}
    \end{itemize}
\end{frame}

\section{Conclusion and Future Work}
\begin{frame}{Conclusion and Future Work}
    \begin{columns}[c,onlytextwidth]
        \begin{column}{0.48\textwidth}
            \svg{../images/cm_ota.svg}{Current Mirror OTA}{1}
        \end{column}
        \begin{column}{0.48\textwidth}
            Both designs perform well across almost all specified performance targets, with the exception of the output-referred noise specification at higher frequencies. Meeting the noise specification at higher frequencies would require increasing the input-transistor bias current, as thermal noise dominates in this regime. However, increasing the bias current in both topologies reduces the output impedance, thereby decreasing loop gain and PSRR performance. Alternative architectures with separate input and gain stages, each optimized for different operating currents, warrant further investigation.
        \end{column}
    \end{columns}
\end{frame}
